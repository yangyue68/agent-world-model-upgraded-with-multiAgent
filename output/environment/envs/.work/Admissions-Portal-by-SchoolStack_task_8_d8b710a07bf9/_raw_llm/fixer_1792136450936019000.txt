```python
import os
import sqlite3
from datetime import datetime
from typing import Optional
from fastapi import FastAPI, HTTPException, Depends
from pydantic import BaseModel
from sqlalchemy import create_engine, Column, Integer, String, DateTime, Text, ForeignKey, CheckConstraint, func
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, Session

# Database setup
DATABASE_URL = os.getenv("DATABASE_PATH")
if not DATABASE_URL:
    raise ValueError("DATABASE_PATH environment variable not set")

engine = create_engine(DATABASE_URL)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

# Models
class User(Base):
    __tablename__ = "users"
    id = Column(Integer, primary_key=True, index=True)
    full_name = Column(String, nullable=False)
    email = Column(String, unique=True, nullable=False)
    role = Column(String, default="staff")
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())

class Program(Base):
    __tablename__ = "programs"
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String, nullable=False)
    degree_type = Column(String, nullable=False)
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())

class Intake(Base):
    __tablename__ = "intakes"
    id = Column(Integer, primary_key=True, index=True)
    program_id = Column(Integer, ForeignKey("programs.id"))
    term = Column(String, nullable=False)
    year = Column(Integer, nullable=False)
    start_date = Column(String)
    end_date = Column(String)
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())

class Applicant(Base):
    __tablename__ = "applicants"
    id = Column(Integer, primary_key=True, index=True)
    first_name = Column(String, nullable=False)
    last_name = Column(String, nullable=False)
    email = Column(String, unique=True, nullable=False)
    phone = Column(String)
    date_of_birth = Column(String)
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())

class Application(Base):
    __tablename__ = "applications"
    id = Column(Integer, primary_key=True, index=True)
    application_number = Column(String, unique=True, nullable=False)
    applicant_id = Column(Integer, ForeignKey("applicants.id"))
    intake_id = Column(Integer, ForeignKey("intakes.id"))
    status = Column(String, nullable=False, default="Submitted",
                   server_default="Submitted")
    submitted_at = Column(DateTime)
    decision_at = Column(DateTime)
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())

    __table_args__ = (
        CheckConstraint("status IN ('Submitted', 'Under Review', 'Interview', 'Decision Pending', 'Admitted', 'Denied', 'Waitlisted', 'Withdrawn')", name='chk_application_status'),
    )

class ApplicationNote(Base):
    __tablename__ = "application_notes"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"))
    author_user_id = Column(Integer, ForeignKey("users.id"))
    note = Column(Text, nullable=False)
    is_internal = Column(Integer, default=1)
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())

    __table_args__ = (
        CheckConstraint("is_internal IN (0, 1)", name='chk_application_note_is_internal'),
    )

class Document(Base):
    __tablename__ = "documents"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"))
    document_type = Column(String)
    file_name = Column(String, nullable=False)
    file_path = Column(String, nullable=False)
    mime_type = Column(String)
    file_size_bytes = Column(Integer)
    uploaded_by_user_id = Column(Integer, ForeignKey("users.id"))
    uploaded_at = Column(DateTime, default=func.current_timestamp())
    is_active = Column(Integer, default=1)
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())

    __table_args__ = (
        CheckConstraint("document_type IN ('Transcript', 'Recommendation Letter', 'Personal Statement', 'Resume', 'Test Score', 'Other')", name='chk_document_type'),
        CheckConstraint("is_active IN (0, 1)", name='chk_document_is_active'),
    )

class EmailLog(Base):
    __tablename__ = "email_log"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"))
    recipient_email = Column(String, nullable=False)
    subject = Column(String, nullable=False)
    body = Column(Text)
    template_name = Column(String)
    sent_by_user_id = Column(Integer, ForeignKey("users.id"))
    sent_at = Column(DateTime, default=func.current_timestamp())
    status = Column(String, default="Sent")
    error_message = Column(String)
    created_at = Column(DateTime, default=func.current_timestamp())

    __table_args__ = (
        CheckConstraint("status IN ('Sent', 'Failed', 'Queued')", name='chk_email_log_status'),
    )

class ApplicationStatusHistory(Base):
    __tablename__ = "application_status_history"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"))
    previous_status = Column(String)
    new_status = Column(String, nullable=False)
    changed_by_user_id = Column(Integer, ForeignKey("users.id"))
    changed_at = Column(DateTime, default=func.current_timestamp())
    reason = Column(Text)

# Request models
class WithdrawApplicationRequest(BaseModel):
    note_text: Optional[str] = None
    reason: Optional[str] = None

class AddApplicationNoteRequest(BaseModel):
    note: str
    is_internal: bool = True

class LogWithdrawalEmailRequest(BaseModel):
    recipient_email: str
    subject: str
    body: Optional[str] = None
    template_name: Optional[str] = None

# Response models
class WithdrawApplicationResponse(BaseModel):
    success: bool
    new_status: str
    application_id: int

class AddApplicationNoteResponse(BaseModel):
    id: int
    created_at: str

class LogWithdrawalEmailResponse(BaseModel):
    id: int
    status: str

class ApplicationDetailsResponse(BaseModel):
    id: int
    application_number: str
    applicant_id: int
    intake_id: int
    status: str
    submitted_at: Optional[str]

class ApplicantDetailsResponse(BaseModel):
    id: int
    first_name: str
    last_name: str
    email: str

# FastAPI app
app = FastAPI()

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@app.get("/applications/{application_number}", response_model=ApplicationDetailsResponse, operation_id="get_application_details")
def get_application_details(application_number: str, db: Session = Depends(get_db)):
    application = db.query(Application).filter(Application.application_number == application_number).first()
    if not application:
        raise HTTPException(status_code=404, detail="Application not found")
    
    return ApplicationDetailsResponse(
        id=application.id,
        application_number=application.application_number,
        applicant_id=application.applicant_id,
        intake_id=application.intake_id,
        status=application.status,
        submitted_at=application.submitted_at.isoformat() if application.submitted_at else None
    )

@app.patch("/applications/{application_number}/withdraw", response_model=WithdrawApplicationResponse, operation_id="withdraw_application")
def withdraw_application(
    application_number: str,
    request: WithdrawApplicationRequest,
    db: Session = Depends(get_db)
):
    # Get the application
    application = db.query(Application).filter(Application.application_number == application_number).first()
    if not application:
        raise HTTPException(status_code=404, detail="Application not found")
    
    # Store old status
    old_status = application.status
    
    # Update application status to Withdrawn
    application.status = "Withdrawn"
    application.updated_at = func.current_timestamp()
    db.commit()
    db.refresh(application)
    
    # Create status history entry
    status_history = ApplicationStatusHistory(
        application_id=application.id,
        previous_status=old_status,
        new_status="Withdrawn",
        changed_by_user_id=1,  # Using current user ID from runtime context
        reason=request.reason if request.reason else "Applicant requested withdrawal"
    )
    db.add(status_history)
    db.commit()
    db.refresh(status_history)
    
    # Add note if provided
    if request.note_text:
        note = ApplicationNote(
            application_id=application.id,
            author_user_id=1,
            note=request.note_text,
            is_internal=True
        )
        db.add(note)
        db.commit()
        db.refresh(note)
    
    return WithdrawApplicationResponse(
        success=True,
        new_status="Withdrawn",
        application_id=application.id
    )

@app.post("/applications/{application_number}/notes", response_model=AddApplicationNoteResponse, operation_id="add_application_note")
def add_application_note(
    application_number: str,
    request: AddApplicationNoteRequest,
    db: Session = Depends(get_db)
):
    # Get the application
    application = db.query(Application).filter(Application.application_number == application_number).first()
    if not application:
        raise HTTPException(status_code=404, detail="Application not found")
    
    # Create note
    note = ApplicationNote(
        application_id=application.id,
        author_user_id=1,  # Using current user ID from runtime context
        note=request.note,
        is_internal=request.is_internal
    )
    db.add(note)
    db.commit()
    db.refresh(note)
    
    return AddApplicationNoteResponse(
        id=note.id,
        created_at=note.created_at.isoformat() if note.created_at else None
    )

@app.post("/applications/{application_number}/emails/log", response_model=LogWithdrawalEmailResponse, operation_id="log_withdrawal_email")
def log_withdrawal_email(
    application_number: str,
    request: LogWithdrawalEmailRequest,
    db: Session = Depends(get_db)
):
    # Get the application
    application = db.query(Application).filter(Application.application_number == application_number).first()
    if not application:
        raise HTTPException(status_code=404, detail="Application not found")
    
    # Create email log entry
    email_log = EmailLog(
        application_id=application.id,
        recipient_email=request.recipient_email,
        subject=request.subject,
        body=request.body,
        template_name=request.template_name,
        sent_by_user_id=1,  # Using current user ID from runtime context
        status="Sent"
    )
    db.add(email_log)
    db.commit()
    db.refresh(email_log)
    
    return LogWithdrawalEmailResponse(
        id=email_log.id,
        status=email_log.status
    )

@app.get("/applicants/{applicant_id}", response_model=ApplicantDetailsResponse, operation_id="get_applicant_details")
def get_applicant_details(applicant_id: int, db: Session = Depends(get_db)):
    applicant = db.query(Applicant).filter(Applicant.id == applicant_id).first()
    if not applicant:
        raise HTTPException(status_code=404, detail="Applicant not found")
    
    return ApplicantDetailsResponse(
        id=applicant.id,
        first_name=applicant.first_name,
        last_name=applicant.last_name,
        email=applicant.email
    )
```
