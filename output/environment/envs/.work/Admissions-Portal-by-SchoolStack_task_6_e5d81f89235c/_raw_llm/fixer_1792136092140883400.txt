import os
import sqlite3
from datetime import datetime, date
from typing import List, Optional
from fastapi import FastAPI, HTTPException, Depends
from pydantic import BaseModel
from sqlalchemy import create_engine, Column, Integer, String, DateTime, Text, CheckConstraint, ForeignKey, func, text
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, Session

# Database setup
DATABASE_URL = os.getenv("DATABASE_PATH")
if not DATABASE_URL:
    raise ValueError("DATABASE_PATH environment variable not set")

engine = create_engine(DATABASE_URL)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

# Models
class User(Base):
    __tablename__ = "users"
    id = Column(Integer, primary_key=True, index=True)
    username = Column(String, unique=True, nullable=False)
    email = Column(String, unique=True, nullable=False)
    full_name = Column(String)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class Program(Base):
    __tablename__ = "programs"
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String, nullable=False)
    code = Column(String, unique=True, nullable=False)
    description = Column(Text)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class Application(Base):
    __tablename__ = "applications"
    id = Column(Integer, primary_key=True, index=True)
    program_id = Column(Integer, ForeignKey("programs.id"), nullable=False)
    applicant_user_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    status = Column(String, nullable=False, default="Draft")
    submitted_at = Column(DateTime)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class RecommendationRequirement(Base):
    __tablename__ = "recommendation_requirements"
    id = Column(Integer, primary_key=True, index=True)
    program_id = Column(Integer, ForeignKey("programs.id"), nullable=False)
    requirement_type = Column(String, nullable=False)
    required_count = Column(Integer, nullable=False, default=1)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class RecommendationRequest(Base):
    __tablename__ = "recommendation_requests"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"), nullable=False)
    requirement_id = Column(Integer, ForeignKey("recommendation_requirements.id"), nullable=False)
    recommender_name = Column(String, nullable=False)
    recommender_email = Column(String, nullable=False)
    recommender_type = Column(String, nullable=False)
    status = Column(String, nullable=False, default="Pending")
    requested_at = Column(DateTime, default=datetime.utcnow)
    submitted_at = Column(DateTime)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class DeadlineRule(Base):
    __tablename__ = "deadline_rules"
    id = Column(Integer, primary_key=True, index=True)
    program_id = Column(Integer, ForeignKey("programs.id"), nullable=False)
    rule_name = Column(String, nullable=False)
    deadline_type = Column(String, nullable=False)
    deadline_date = Column(DateTime, nullable=False)
    action_on_missing = Column(String, nullable=False)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class ReminderSchedule(Base):
    __tablename__ = "reminder_schedules"
    id = Column(Integer, primary_key=True, index=True)
    program_id = Column(Integer, ForeignKey("programs.id"), nullable=False)
    reminder_type = Column(String, nullable=False)
    interval_days = Column(Integer, nullable=False)
    start_date = Column(DateTime, nullable=False)
    end_date = Column(DateTime)
    is_active = Column(Integer, nullable=False, default=1)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class ReminderLog(Base):
    __tablename__ = "reminder_logs"
    id = Column(Integer, primary_key=True, index=True)
    reminder_schedule_id = Column(Integer, ForeignKey("reminder_schedules.id"), nullable=False)
    application_id = Column(Integer, ForeignKey("applications.id"), nullable=False)
    sent_at = Column(DateTime, default=datetime.utcnow)
    status = Column(String, nullable=False, default="Sent")

# Request models
class LookupProgramsQuery(BaseModel):

    required_count: int

    id: int
    program_id: int
    rule_name: str

# FastAPI app
app = FastAPI()

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@app.get("/programs/lookup", response_model=List[ProgramResponse], operation_id="lookup_programs")
def lookup_programs(name: Optional[str] = None, code: Optional[str] = None, db: Session = Depends(get_db)):
    query = db.query(Program)
    if name:
        query = query.filter(Program.name.contains(name))
    if code:
        query = query.filter(Program.code == code)
    return query.all()

@app.post("/programs/{program_id}/recommendation-requirements", response_model=RecommendationRequirementResponse, operation_id="create_recommendation_requirement")
def create_recommendation_requirement(
    program_id: int,
    requirement: CreateRecommendationRequirement,
    db: Session = Depends(get_db)
):
    # Check if program exists
    program = db.query(Program).filter(Program.id == program_id).first()
    if not program:
        raise HTTPException(status_code=404, detail="Program not found")
    
    # Check if requirement already exists for this type
    existing_req = db.query(RecommendationRequirement).filter(
        RecommendationRequirement.program_id == program_id,
        RecommendationRequirement.requirement_type == requirement.requirement_type
    ).first()
    
    if existing_req:
        # Update existing requirement
        existing_req.required_count = requirement.required_count
        existing_req.updated_at = datetime.utcnow()
        db.commit()
        db.refresh(existing_req)
        return existing_req
    else:
        # Create new requirement
        new_req = RecommendationRequirement(
            program_id=program_id,
            requirement_type=requirement.requirement_type,
            required_count=requirement.required_count
        )
        db.add(new_req)
        db.commit()
        db.refresh(new_req)
        return new_req

@app.post("/programs/{program_id}/deadline-rules", response_model=DeadlineRuleResponse, operation_id="create_deadline_rule")
def create_deadline_rule(
    program_id: int,
    rule: CreateDeadlineRule,
    db: Session = Depends(get_db)
):
    # Check if program exists
    program = db.query(Program).filter(Program.id == program_id).first()
    if not program:
        raise HTTPException(status_code=404, detail="Program not found")
    
    # Create new deadline rule
    new_rule = DeadlineRule(
        program_id=program_id,
        rule_name=rule.rule_name,
        deadline_type=rule.deadline_type,
        deadline_date=datetime.fromisoformat(rule.deadline_date),
        action_on_missing=rule.action_on_missing
    )
    db.add(new_rule)
    db.commit()
    db.refresh(new_rule)
    return new_rule

@app.post("/programs/{program_id}/reminder-schedules", response_model=ReminderScheduleResponse, operation_id="create_reminder_schedule")
def create_reminder_schedule(
    program_id: int,
    reminder: CreateReminderSchedule,
    db: Session = Depends(get_db)
):
    # Check if program exists
    program = db.query(Program).filter(Program.id == program_id).first()
    if not program:
        raise HTTPException(status_code=404, detail="Program not found")
    
    # Create new reminder schedule
    new_reminder = ReminderSchedule(
        program_id=program_id,
        reminder_type=reminder.reminder_type,
        interval_days=reminder.interval_days,
        start_date=datetime.fromisoformat(reminder.start_date),
        end_date=datetime.fromisoformat(reminder.end_date) if reminder.end_date else None,
        is_active=1
    )
    db.add(new_reminder)
    db.commit()
    db.refresh(new_reminder)
    return new_reminder

@app.on_event("startup")
def configure_md_program():
    db = SessionLocal()
    try:
        # Find the MD Program 2026
        md_program = db.query(Program).filter(Program.code == "MD-2026").first()
        if not md_program:
            raise HTTPException(status_code=404, detail="MD Program 2026 not found")
        
        # Create recommendation requirements
        # Two science faculty recommendations
        create_recommendation_requirement(md_program.id, CreateRecommendationRequirement(
            requirement_type="Science Faculty",
            required_count=2
        ), db)
        
        # One clinical supervisor recommendation
        create_recommendation_requirement(md_program.id, CreateRecommendationRequirement(
            requirement_type="Clinical Supervisor",
            required_count=1
        ), db)
        
        # Set recommendation submission deadline to December 15, 2025
        create_deadline_rule(md_program.id, CreateDeadlineRule(
            rule_name="MD Rec Deadline",
            deadline_type="Recommendation Submission",
            deadline_date="2025-12-15T23:59:00",
            action_on_missing="Waitlist Review"
        ), db)
        
        # Enable automated reminder emails every 7 days starting December 1, 2025
        create_reminder_schedule(md_program.id, CreateReminderSchedule(
            reminder_type="Recommendation Incomplete",
            interval_days=7,
            start_date="2025-12-01T00:00:00",
            end_date="2025-12-15T23:59:00"
        ), db)
        
        # Create deadline rule that moves applications to Waitlist Review after Jan 10, 2026
        create_deadline_rule(md_program.id, CreateDeadlineRule(
            rule_name="MD Final Application Deadline",
            deadline_type="Final Application",
            deadline_date="2026-01-10T23:59:00",
            action_on_missing="Waitlist Review"
        ), db)
        
    except Exception as e:
        print(f"Error during configuration: {str(e)}")
        raise
    finally:
        db.close()
