import os
import sqlite3
from datetime import date
from typing import List, Optional
from fastapi import FastAPI, HTTPException, Depends
from pydantic import BaseModel
from sqlalchemy import create_engine, Column, Integer, String, Boolean, Date, ForeignKey, Text, CheckConstraint, text
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, Session

# Database setup
DATABASE_URL = os.getenv("DATABASE_PATH")
if not DATABASE_URL:
    raise ValueError("DATABASE_PATH environment variable not set")

engine = create_engine(DATABASE_URL)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

# Models
class User(Base):
    __tablename__ = "users"
    id = Column(Integer, primary_key=True, index=True)
    username = Column(String, unique=True, nullable=False)
    full_name = Column(String)
    email = Column(String, unique=True)
    created_at = Column(Date)
    updated_at = Column(Date)

class Form(Base):
    __tablename__ = "forms"
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String, nullable=False)
    description = Column(Text)
    created_by = Column(Integer, ForeignKey("users.id"))
    created_at = Column(Date)
    updated_at = Column(Date)

class FormSection(Base):
    __tablename__ = "form_sections"
    id = Column(Integer, primary_key=True, index=True)
    form_id = Column(Integer, ForeignKey("forms.id"))
    name = Column(String, nullable=False)
    description = Column(Text)
    is_optional = Column(Boolean, nullable=False, default=False)
    max_entries = Column(Integer)
    display_order = Column(Integer, nullable=False, default=0)
    created_at = Column(Date)
    updated_at = Column(Date)

class SectionField(Base):
    __tablename__ = "section_fields"
    id = Column(Integer, primary_key=True, index=True)
    section_id = Column(Integer, ForeignKey("form_sections.id"))
    field_name = Column(String, nullable=False)
    field_label = Column(String, nullable=False)
    field_type = Column(String, nullable=False)
    is_required = Column(Boolean, nullable=False, default=False)
    display_order = Column(Integer, nullable=False, default=0)
    created_at = Column(Date)
    updated_at = Column(Date)

class FormQuestion(Base):
    __tablename__ = "form_questions"
    id = Column(Integer, primary_key=True, index=True)
    form_id = Column(Integer, ForeignKey("forms.id"))
    question_text = Column(Text, nullable=False)
    question_type = Column(String, nullable=False)
    is_required = Column(Boolean, nullable=False, default=False)
    display_order = Column(Integer, nullable=False, default=0)
    created_at = Column(Date)
    updated_at = Column(Date)

class FormDocument(Base):
    __tablename__ = "form_documents"
    id = Column(Integer, primary_key=True, index=True)
    form_id = Column(Integer, ForeignKey("forms.id"))
    name = Column(String, nullable=False)
    description = Column(Text)
    is_required = Column(Boolean, nullable=False, default=False)
    condition_question_id = Column(Integer, ForeignKey("form_questions.id"))
    condition_value = Column(String)
    created_at = Column(Date)
    updated_at = Column(Date)

class FormDeadline(Base):
    __tablename__ = "form_deadlines"
    id = Column(Integer, primary_key=True, index=True)
    form_id = Column(Integer, ForeignKey("forms.id"))
    deadline_type = Column(String, nullable=False)
    deadline_date = Column(Date)
    created_at = Column(Date)
    updated_at = Column(Date)

class Application(Base):
    __tablename__ = "applications"
    id = Column(Integer, primary_key=True, index=True)
    form_id = Column(Integer, ForeignKey("forms.id"))
    applicant_id = Column(Integer, ForeignKey("users.id"))
    status = Column(String, nullable=False, default="draft")
    submitted_at = Column(Date)
    created_at = Column(Date)
    updated_at = Column(Date)

class ApplicationAnswer(Base):
    __tablename__ = "application_answers"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"))
    question_id = Column(Integer, ForeignKey("form_questions.id"))
    answer_value = Column(Text)
    created_at = Column(Date)
    updated_at = Column(Date)

class ApplicationSectionEntry(Base):
    __tablename__ = "application_section_entries"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"))
    section_id = Column(Integer, ForeignKey("form_sections.id"))
    entry_index = Column(Integer, nullable=False)
    created_at = Column(Date)
    updated_at = Column(Date)

class ApplicationSectionEntryValue(Base):
    __tablename__ = "application_section_entry_values"
    id = Column(Integer, primary_key=True, index=True)
    entry_id = Column(Integer, ForeignKey("application_section_entries.id"))
    field_id = Column(Integer, ForeignKey("section_fields.id"))
    value = Column(Text)
    created_at = Column(Date)
    updated_at = Column(Date)

class ApplicationDocument(Base):
    __tablename__ = "application_documents"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"))
    document_id = Column(Integer, ForeignKey("form_documents.id"))
    file_path = Column(Text, nullable=False)
    uploaded_at = Column(Date)
    created_at = Column(Date)
    updated_at = Column(Date)

# Pydantic models

    form_id: int
    name: str
    is_optional: bool
    max_entries: Optional[int]

class CreateSectionFieldResponse(BaseModel):

# FastAPI app
app = FastAPI()

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@app.get("/forms", response_model=List[FormResponse])
def find_forms_by_name(name: str, db: Session = Depends(get_db)):
    forms = db.query(Form).filter(Form.name.like(f"%{name}%")).all()
    return [{"id": f.id, "name": f.name} for f in forms]

@app.post("/forms/{form_id}/sections", response_model=CreateFormSectionResponse)
def create_form_section(form_id: int, section_data: CreateFormSectionRequest, db: Session = Depends(get_db)):
    section = FormSection(
        form_id=form_id,
        name=section_data.name,
        description=section_data.description,
        is_optional=section_data.is_optional,
        max_entries=section_data.max_entries,
        display_order=section_data.display_order or 0
    )
    db.add(section)
    db.commit()
    db.refresh(section)
    return {
        "id": section.id,
        "form_id": section.form_id,
        "name": section.name,
        "is_optional": section.is_optional,
        "max_entries": section.max_entries
    }

@app.post("/sections/{section_id}/fields", response_model=CreateSectionFieldResponse)
def create_section_field(section_id: int, field_data: CreateSectionFieldRequest, db: Session = Depends(get_db)):
    field = SectionField(
        section_id=section_id,
        field_name=field_data.field_name,
        field_label=field_data.field_label,
        field_type=field_data.field_type,
        is_required=field_data.is_required,
        display_order=field_data.display_order or 0
    )
    db.add(field)
    db.commit()
    db.refresh(field)
    return {
        "id": field.id,
        "section_id": field.section_id,
        "field_name": field.field_name,
        "field_label": field.field_label,
        "field_type": field.field_type,
        "is_required": field.is_required
    }

@app.get("/forms/{form_id}/questions", response_model=List[FormQuestionResponse])
def get_form_questions(form_id: int, db: Session = Depends(get_db)):
    questions = db.query(FormQuestion).filter(FormQuestion.form_id == form_id).all()
    return [
        {
            "id": q.id,
            "question_text": q.question_text,
            "question_type": q.question_type,
            "is_required": q.is_required
        } for q in questions
    ]

@app.post("/forms/{form_id}/documents", response_model=CreateFormDocumentResponse)

@app.get("/forms/{form_id}/deadlines", response_model=List[FormDeadlineResponse])
def get_form_deadlines(form_id: int, db: Session = Depends(get_db)):
    deadlines = db.query(FormDeadline).filter(FormDeadline.form_id == form_id).all()
    return [
        {
            "id": d.id,
            "deadline_type": d.deadline_type,
            "deadline_date": d.deadline_date
        } for d in deadlines
    ]

@app.put("/deadlines/{deadline_id}", response_model=UpdateFormDeadlineResponse)
def update_form_deadline(deadline_id: int, deadline_data: UpdateFormDeadlineRequest, db: Session = Depends(get_db)):
    deadline = db.query(FormDeadline).filter(FormDeadline.id == deadline_id).first()
    if not deadline:
        raise HTTPException(status_code=404, detail="Deadline not found")
    
    deadline.deadline_date = deadline_data.deadline_date
    db.commit()
    db.refresh(deadline)
    return {
        "id": deadline.id,
        "form_id": deadline.form_id,
        "deadline_type": deadline.deadline_type,
        "deadline_date": deadline.deadline_date
    }

# Task implementation
@app.post("/task_5", status_code=200)
def execute_task_5(db: Session = Depends(get_db)):
    # Find the form
    form = db.query(Form).filter(Form.name == "Fall 2025 Undergraduate Admissions").first()
    if not form:
        raise HTTPException(status_code=404, detail="Form not found")

    # Step 1: Add 'Extracurricular Activities' section
    extracurricular_section = FormSection(
        form_id=form.id,
        name="Extracurricular Activities",
        description="Details about your extracurricular activities",
        is_optional=True,
        max_entries=5,
        display_order=3
    )
    db.add(extracurricular_section)
    db.flush()  # Get the ID without committing yet
    
    # Step 2: Add fields to the section
    fields_data = [
        {"field_name": "activity_name", "field_label": "Activity Name", "field_type": "text", "is_required": False},
        {"field_name": "role", "field_label": "Role", "field_type": "text", "is_required": False},
        {"field_name": "hours_per_week", "field_label": "Hours per Week", "field_type": "number", "is_required": False},
        {"field_name": "description", "field_label": "Description", "field_type": "textarea", "is_required": False}
    ]
    
    for i, field_data in enumerate(fields_data):
        field = SectionField(
            section_id=extracurricular_section.id,
            field_name=field_data["field_name"],
            field_label=field_data["field_label"],
            field_type=field_data["field_type"],
            is_required=field_data["is_required"],
            display_order=i + 1
        )
        db.add(field)
    
    # Step 3: Find the question ID for "Are you applying for need-based aid?"
    need_based_question = db.query(FormQuestion).filter(
        FormQuestion.form_id == form.id,
        FormQuestion.question_text == "Are you applying for need-based aid?"
    ).first()
    
    if not need_based_question:
        raise HTTPException(status_code=404, detail="Need-based aid question not found")
    
    # Step 4: Add 'Financial Aid Statement' document
    financial_aid_document = FormDocument(
        form_id=form.id,
        name="Financial Aid Statement",
        description="Required statement for need-based aid applicants",
        is_required=True,
        condition_question_id=need_based_question.id,
        condition_value="Yes"
    )
    db.add(financial_aid_document)
    
    # Step 5: Update regular decision deadline
    regular_decision_deadline = db.query(FormDeadline).filter(
        FormDeadline.form_id == form.id,
        FormDeadline.deadline_type == "regular_decision"
    ).first()
    
    if not regular_decision_deadline:
        raise HTTPException(status_code=404, detail="Regular decision deadline not found")
    
    regular_decision_deadline.deadline_date = date(2025, 1, 5)
    
    # Commit all changes
    db.commit()
    
    return {"message": "Task 5 executed successfully"}
