```python
import re
from datetime import datetime
from typing import Optional
from urllib.parse import urlparse

from fastapi import FastAPI, HTTPException, Path
from pydantic import BaseModel
from sqlalchemy import (
    Column,
    DateTime,
    ForeignKey,
    Integer,
    String,
    Text,
    create_engine,
    func,
    select,
    text,
)
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker

# Database setup
DATABASE_PATH = "sqlite:///./test.db"
engine = create_engine(DATABASE_PATH)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

# Models
class Applicant(Base):
    __tablename__ = "applicants"
    id = Column(Integer, primary_key=True, index=True)
    first_name = Column(String, nullable=False)
    last_name = Column(String, nullable=False)
    email = Column(String, unique=True, nullable=False)
    phone = Column(String)
    date_of_birth = Column(String)
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())


class Application(Base):
    __tablename__ = "applications"
    id = Column(Integer, primary_key=True, index=True)
    application_number = Column(String, unique=True, nullable=False)
    applicant_id = Column(Integer, ForeignKey("applicants.id"), nullable=False)
    intake_id = Column(Integer, ForeignKey("intakes.id"), nullable=False)
    status = Column(
        String,
        nullable=False,
        default="Submitted",
        server_default="Submitted",
        comment="CHECK (status IN ('Submitted', 'Under Review', 'Interview', 'Decision Pending', 'Admitted', 'Denied', 'Waitlisted', 'Withdrawn'))",
    )
    submitted_at = Column(DateTime)
    decision_at = Column(DateTime)
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())


class ApplicationNote(Base):
    __tablename__ = "application_notes"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"), nullable=False)
    author_user_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    note = Column(Text, nullable=False)
    is_internal = Column(Integer, nullable=False, default=1, comment="CHECK (is_internal IN (0, 1))")
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())


class EmailLog(Base):
    __tablename__ = "email_log"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"))
    recipient_email = Column(String, nullable=False)
    subject = Column(String, nullable=False)
    body = Column(Text)
    template_name = Column(String)
    sent_by_user_id = Column(Integer, ForeignKey("users.id"))
    sent_at = Column(DateTime, default=func.current_timestamp())
    status = Column(String, nullable=False, default="Sent", comment="CHECK (status IN ('Sent', 'Failed', 'Queued'))")
    error_message = Column(Text)
    created_at = Column(DateTime, default=func.current_timestamp())


class ApplicationStatusHistory(Base):
    __tablename__ = "application_status_history"
    id = Column(Integer, primary_key=True, index=True)
    application_id = Column(Integer, ForeignKey("applications.id"), nullable=False)
    previous_status = Column(String)
    new_status = Column(String, nullable=False)
    changed_by_user_id = Column(Integer, ForeignKey("users.id"))
    changed_at = Column(DateTime, default=func.current_timestamp())
    reason = Column(Text)


class User(Base):
    __tablename__ = "users"
    id = Column(Integer, primary_key=True, index=True)
    full_name = Column(String, nullable=False)
    email = Column(String, unique=True, nullable=False)
    role = Column(String, nullable=False, default="staff")
    created_at = Column(DateTime, default=func.current_timestamp())
    updated_at = Column(DateTime, default=func.current_timestamp(), onupdate=func.current_timestamp())


# Pydantic models
class ApplicationDetails(BaseModel):
    id: int
    application_number: str
    applicant_id: int
    intake_id: int
    status: str
    submitted_at: Optional[str]

    class Config:
        orm_mode = True


class WithdrawApplicationResponse(BaseModel):
    success: bool
    new_status: str
    application_id: int

    class Config:
        orm_mode = True


class AddApplicationNoteResponse(BaseModel):
    id: int
    created_at: str

    class Config:
        orm_mode = True


class LogWithdrawalEmailResponse(BaseModel):
    id: int
    status: str

    class Config:
        orm_mode = True


class ApplicantDetails(BaseModel):
    id: int
    first_name: str
    last_name: str
    email: str

    class Config:
        orm_mode = True


# FastAPI app
app = FastAPI()


def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


@app.get("/applications/{application_number}", response_model=ApplicationDetails)
def get_application_details(
    application_number: str = Path(..., description="The unique application number (e.g., APP-3172)")
):
    db = next(get_db())
    application = db.execute(
        select(Application).where(Application.application_number == application_number)
    ).scalar_one_or_none()
    if not application:
        raise HTTPException(status_code=404, detail="Application not found")
    return application


@app.patch("/applications/{application_number}/withdraw", response_model=WithdrawApplicationResponse)
def withdraw_application(
    application_number: str = Path(..., description="The unique application number to withdraw"),
    note_text: Optional[str] = None,
    reason: Optional[str] = None,
):
    db = next(get_db())
    
    # Get application
    application = db.execute(
        select(Application).where(Application.application_number == application_number)
    ).scalar_one_or_none()
    if not application:
        raise HTTPException(status_code=404, detail="Application not found")

    # Record status change
    old_status = application.status
    application.status = "Withdrawn"
    application.updated_at = func.current_timestamp()
    db.add(application)

    # Create status history entry
    status_history = ApplicationStatusHistory(
        application_id=application.id,
        previous_status=old_status,
        new_status="Withdrawn",
        reason=reason,
    )
    db.add(status_history)

    # Add note if provided
    if note_text:
        note = ApplicationNote(
            application_id=application.id,
            author_user_id=2,  # Default user ID for staff
            note=note_text,
            is_internal=True,
        )
        db.add(note)

    db.commit()
    db.refresh(application)

    return WithdrawApplicationResponse(
        success=True,
        new_status=application.status,
        application_id=application.id,
    )


@app.post("/applications/{application_number}/notes", response_model=AddApplicationNoteResponse)
def add_application_note(
    application_number: str = Path(..., description="The unique application number"),
    note: str = "",
    is_internal: bool = True,
):
    db = next(get_db())
    
    # Get application
    application = db.execute(
        select(Application).where(Application.application_number == application_number)
    ).scalar_one_or_none()
    if not application:
        raise HTTPException(status_code=404, detail="Application not found")

    # Create note
    note_obj = ApplicationNote(
        application_id=application.id,
        author_user_id=2,  # Default user ID for staff
        note=note,
        is_internal=int(is_internal),
    )
    db.add(note_obj)
    db.commit()
    db.refresh(note_obj)

    return AddApplicationNoteResponse(
        id=note_obj.id,
        created_at=note_obj.created_at.isoformat() if note_obj.created_at else None,
    )


@app.post("/applications/{application_number}/emails/log", response_model=LogWithdrawalEmailResponse)
def log_withdrawal_email(
    application_number: str = Path(..., description="The unique application number"),
    recipient_email: str = "",
    subject: str = "",
    body: Optional[str] = None,
    template_name: Optional[str] = None,
):
    db = next(get_db())
    
    # Get application
    application = db.execute(
        select(Application).where(Application.application_number == application_number)
    ).scalar_one_or_none()
    if not application:
        raise HTTPException(status_code=404, detail="Application not found")

    # Create email log entry
    email_log = EmailLog(
        application_id=application.id,
        recipient_email=recipient_email,
        subject=subject,
        body=body,
        template_name=template_name,
        sent_by_user_id=2,  # Default user ID for staff
        status="Sent",
    )
    db.add(email_log)
    db.commit()
    db.refresh(email_log)

    return LogWithdrawalEmailResponse(
        id=email_log.id,
        status=email_log.status,
    )


@app.get("/applicants/{applicant_id}", response_model=ApplicantDetails)
def get_applicant_details(applicant_id: int = Path(..., description="The ID of the applicant")):
    db = next(get_db())
    applicant = db.execute(select(Applicant).where(Applicant.id == applicant_id)).scalar_one_or_none()
    if not applicant:
        raise HTTPException(status_code=404, detail="Applicant not found")
    return applicant


# Main function to run the application
if __name__ == "__main__":
    import uvicorn

    Base.metadata.create_all(bind=engine)
    uvicorn.run(app, host="0.0.0.0", port=8000)
```
