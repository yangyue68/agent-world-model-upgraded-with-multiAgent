```python
import os
import sqlite3
from datetime import datetime
from typing import List, Optional
from fastapi import FastAPI, HTTPException, Path, Query
from pydantic import BaseModel
from sqlalchemy import create_engine, Column, Integer, String, Text, DateTime, ForeignKey, CheckConstraint, func, text
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, relationship

# Database setup
DATABASE_URL = os.getenv("DATABASE_PATH")
if not DATABASE_URL:
    raise ValueError("DATABASE_PATH environment variable not set")

engine = create_engine(DATABASE_URL)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

# Models
class User(Base):
    __tablename__ = "users"
    id = Column(Integer, primary_key=True, index=True)
    username = Column(String, unique=True, nullable=False)
    full_name = Column(String)
    email = Column(String, unique=True)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class Program(Base):
    __tablename__ = "programs"
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String, nullable=False)
    term = Column(String, nullable=False)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    __table_args__ = (CheckConstraint(sqltext="name IS NOT NULL"), CheckConstraint(sqltext="term IS NOT NULL"),)

class Applicant(Base):
    __tablename__ = "applicants"
    id = Column(String, primary_key=True, index=True)
    program_id = Column(Integer, ForeignKey("programs.id"), nullable=False)
    first_name = Column(String, nullable=False)
    last_name = Column(String, nullable=False)
    date_of_birth = Column(String)
    email = Column(String)
    phone = Column(String)
    status = Column(String, nullable=False, default="active")
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    __table_args__ = (CheckConstraint(sqltext="status IN ('active', 'merged', 'deleted')"),)

class Document(Base):
    __tablename__ = "documents"
    id = Column(Integer, primary_key=True, index=True)
    applicant_id = Column(String, ForeignKey("applicants.id"), nullable=False)
    file_name = Column(String, nullable=False)
    file_path = Column(String, nullable=False)
    file_type = Column(String)
    file_size = Column(Integer)
    uploaded_at = Column(DateTime, default=datetime.utcnow)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class TestScore(Base):
    __tablename__ = "test_scores"
    id = Column(Integer, primary_key=True, index=True)
    applicant_id = Column(String, ForeignKey("applicants.id"), nullable=False)
    test_name = Column(String, nullable=False)
    score = Column(Integer)
    test_date = Column(String)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class RecommendationRequest(Base):
    __tablename__ = "recommendation_requests"
    id = Column(Integer, primary_key=True, index=True)
    applicant_id = Column(String, ForeignKey("applicants.id"), nullable=False)
    recommender_name = Column(String, nullable=False)
    recommender_email = Column(String, nullable=False)
    status = Column(String, nullable=False, default="pending")
    requested_at = Column(DateTime, default=datetime.utcnow)
    submitted_at = Column(DateTime)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    __table_args__ = (CheckConstraint(sqltext="status IN ('pending', 'submitted', 'declined')"),)

class SubmissionResponse(Base):
    __tablename__ = "submission_responses"
    id = Column(Integer, primary_key=True, index=True)
    applicant_id = Column(String, ForeignKey("applicants.id"), nullable=False)
    question_key = Column(String, nullable=False)
    response_text = Column(Text)
    submitted_at = Column(DateTime, default=datetime.utcnow)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class MergeLog(Base):
    __tablename__ = "merge_logs"
    id = Column(Integer, primary_key=True, index=True)
    primary_applicant_id = Column(String, ForeignKey("applicants.id"), nullable=False)
    duplicate_applicant_id = Column(String, ForeignKey("applicants.id"), nullable=False)
    reviewer_user_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    merged_at = Column(DateTime, default=datetime.utcnow)
    details = Column(Text)
    created_at = Column(DateTime, default=datetime.utcnow)

# Request models
class MergeDuplicateApplicantsRequest(BaseModel):
    reviewer_user_id: int

# Response models
class MergeResult(BaseModel):
    success: bool
    message: str
    primary_applicant_id: str
    merged_from_applicant_id: str
    moved_documents_count: int
    moved_test_scores_count: int
    moved_recommendation_requests_count: int
    moved_submission_responses_count: int
    merge_log_id: int

class ApplicantLookupResponse(BaseModel):
    id: str
    first_name: str
    last_name: str
    date_of_birth: Optional[str]
    email: Optional[str]
    phone: Optional[str]
    status: str
    program_id: int

# FastAPI app
app = FastAPI()

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@app.get("/applicants/lookup", response_model=List[ApplicantLookupResponse], operation_id="lookup_applicants_by_name_and_program")
def lookup_applicants_by_name_and_program(
    first_name: str = Query(...),
    last_name: str = Query(...),
    program_name: str = Query(...),
    program_term: str = Query(...),
    db = None
):
    if db is None:
        db = next(get_db())
    
    program = db.query(Program).filter(
        Program.name == program_name,
        Program.term == program_term
    ).first()
    
    if not program:
        raise HTTPException(status_code=404, detail="Program not found")
    
    applicants = db.query(Applicant).filter(
        Applicant.program_id == program.id,
        Applicant.first_name == first_name,
        Applicant.last_name == last_name
    ).all()
    
    return [
        ApplicantLookupResponse(
            id=applicant.id,
            first_name=applicant.first_name,
            last_name=applicant.last_name,
            date_of_birth=applicant.date_of_birth,
            email=applicant.email,
            phone=applicant.phone,
            status=applicant.status,
            program_id=applicant.program_id
        ) for applicant in applicants
    ]

@app.post("/applicants/{primary_id}/merge/{duplicate_id}", response_model=MergeResult, operation_id="merge_duplicate_applicants")
def merge_duplicate_applicants(body: MergeDuplicateApplicantsRequest,
    primary_id: str = Path(...),
    duplicate_id: str = Path(...),
    db = None):
    if db is None:
        db = next(get_db())
    
    # Validate primary and duplicate applicants exist
    primary_applicant = db.query(Applicant).filter(Applicant.id == primary_id).first()
    duplicate_applicant = db.query(Applicant).filter(Applicant.id == duplicate_id).first()
    
    if not primary_applicant:
        raise HTTPException(status_code=404, detail=f"Primary applicant {primary_id} not found")
    
    if not duplicate_applicant:
        raise HTTPException(status_code=404, detail=f"Duplicate applicant {duplicate_id} not found")
    
    # Verify they're in the same program
    if primary_applicant.program_id != duplicate_applicant.program_id:
        raise HTTPException(status_code=400, detail="Applicants must be in the same program")
    
    # Verify primary applicant is active
    if primary_applicant.status != "active":
        raise HTTPException(status_code=400, detail="Primary applicant must be active")
    
    # Verify duplicate applicant is active
    if duplicate_applicant.status != "active":
        raise HTTPException(status_code=400, detail="Duplicate applicant must be active")
    
    # Verify reviewer exists
    reviewer = db.query(User).filter(User.id == body.reviewer_user_id).first()
    if not reviewer:
        raise HTTPException(status_code=404, detail="Reviewer user not found")
    
    # Count items before merge
    documents_count = db.query(Document).filter(Document.applicant_id == duplicate_id).count()
    test_scores_count = db.query(TestScore).filter(TestScore.applicant_id == duplicate_id).count()
    recommendation_requests_count = db.query(RecommendationRequest).filter(RecommendationRequest.applicant_id == duplicate_id).count()
    submission_responses_count = db.query(SubmissionResponse).filter(SubmissionResponse.applicant_id == duplicate_id).count()
    
    # Move documents
    db.execute(
        text("UPDATE documents SET applicant_id = :primary_id WHERE applicant_id = :duplicate_id"),
        {"primary_id": primary_id, "duplicate_id": duplicate_id}
    )
    
    # Move test scores
    db.execute(
        text("UPDATE test_scores SET applicant_id = :primary_id WHERE applicant_id = :duplicate_id"),
        {"primary_id": primary_id, "duplicate_id": duplicate_id}
    )
    
    # Move recommendation requests
    db.execute(
        text("UPDATE recommendation_requests SET applicant_id = :primary_id WHERE applicant_id = :duplicate_id"),
        {"primary_id": primary_id, "duplicate_id": duplicate_id}
    )
    
    # Move submission responses
    db.execute(
        text("UPDATE submission_responses SET applicant_id = :primary_id WHERE applicant_id = :duplicate_id"),
        {"primary_id": primary_id, "duplicate_id": duplicate_id}
    )
    
    # Resolve conflicting date_of_birth - keep from primary
    # Update primary applicant with any missing info from duplicate
    if not primary_applicant.date_of_birth and duplicate_applicant.date_of_birth:
        primary_applicant.date_of_birth = duplicate_applicant.date_of_birth
    
    if not primary_applicant.email and duplicate_applicant.email:
        primary_applicant.email = duplicate_applicant.email
    
    if not primary_applicant.phone and duplicate_applicant.phone:
        primary_applicant.phone = duplicate_applicant.phone
    
    # Mark duplicate as merged
    duplicate_applicant.status = "merged"
    
    # Create merge log
    merge_log = MergeLog(
        primary_applicant_id=primary_id,
        duplicate_applicant_id=duplicate_id,
        reviewer_user_id=body.reviewer_user_id,
        details=f"Merged duplicate applicant {duplicate_id} into primary applicant {primary_id}"
    )
    db.add(merge_log)
    db.commit()
    db.refresh(merge_log)
    
    return MergeResult(
        success=True,
        message=f"Successfully merged duplicate applicant {duplicate_id} into primary applicant {primary_id}",
        primary_applicant_id=primary_id,
        merged_from_applicant_id=duplicate_id,
        moved_documents_count=documents_count,
        moved_test_scores_count=test_scores_count,
        moved_recommendation_requests_count=recommendation_requests_count,
        moved_submission_responses_count=submission_responses_count,
        merge_log_id=merge_log.id
    )

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
```
